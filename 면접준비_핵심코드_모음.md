### WebSocketConfig.java
```java
@Configuration
@EnableWebSocketMessageBroker
public class WebSocketConfig implements WebSocketMessageBrokerConfigurer {
    
    private final WebSocketAuthInterceptor webSocketAuthInterceptor;
    
    @Value("${app.websocket.allowed-origins}")
    private String allowedOrigins;
    
    // STOMP 엔드포인트 등록
    @Override
    public void registerStompEndpoints(StompEndpointRegistry registry) {
        registry.addEndpoint("/ws/chat")
                .setAllowedOrigins(allowedOrigins.split(","))
                .addInterceptors(webSocketAuthInterceptor)  // 인증 인터셉터
                .withSockJS();  // SockJS 폴백 지원
    }
    
    // 메시지 브로커 설정
    @Override
    public void configureMessageBroker(MessageBrokerRegistry registry) {
        // 클라이언트 → 서버
        registry.setApplicationDestinationPrefixes("/app");
        
        // 서버 → 클라이언트 (브로드캐스트)
        registry.enableSimpleBroker("/topic");
    }
}
```

**설명**:
- `/ws/chat`: WebSocket 연결 엔드포인트
- `/app`: 클라이언트가 서버로 메시지 보낼 때 prefix
- `/topic`: 서버가 여러 클라이언트에게 브로드캐스트할 때 prefix

```java
public class ChatService {
    
    private final SimpMessagingTemplate messagingTemplate;
    private final ChatMessageRepository repository;
    
    public void saveAndBroadcastMessage(ChatMessageDTO dto, Long userId) {
        // 1. DB 저장

            .senderId(userId)
            .chatMessageContent(dto.getChatMessageContent())

            "/topic/chat/" + dto.getChatRoomId(),
            convert(saved)
        );
    }
```

## 2️⃣ 채팅 - WebSocket 연결 (Frontend)

### chatSocket.js
```javascript
import SockJS from 'sockjs-client/dist/sockjs.min.js';
import { Client } from "@stomp/stompjs";

const ENDPOINT = `/ws/chat`;
let stompClient = null;
let subscription = null;

export function connectStomp(roomId, onMessage, onConnect, onError) {
  console.log('채팅 연결 시작:', roomId);
  
  stompClient = new Client({
    // SockJS 객체 생성 (쿠키 자동 전송)
    webSocketFactory: () => new SockJS(ENDPOINT),
    
    // 자동 재연결
    reconnectDelay: 5000,
    
    // 연결 성공 콜백
    onConnect: (frame) => {
      console.log('WebSocket 연결 성공');
      
      // 채팅방 구독
      subscription = stompClient.subscribe(
        `/topic/chat/${roomId}`,
        (message) => {
          const parsed = JSON.parse(message.body);
          onMessage(parsed);  // 콜백 실행
        }
      );
      
      onConnect && onConnect();
    },
    
    // 에러 콜백
    onStompError: (error) => {
      console.error('STOMP 에러:', error);
      onError && onError(error);
    }
  });
  
  stompClient.activate();  // 연결 시작
}

// 메시지 전송
export function sendMessage(roomId, content) {
  if (!stompClient || !stompClient.connected) {
    throw new Error('WebSocket이 연결되지 않았습니다');
  }
  
  stompClient.publish({
    destination: '/app/chat/message',
    body: JSON.stringify({
      chatRoomId: roomId,
      chatMessageContent: content
    })
  });
}

// 연결 해제
export function disconnectStomp() {
  if (subscription) {
    subscription.unsubscribe();
    subscription = null;
  }
  
  if (stompClient) {
    stompClient.deactivate();
    stompClient = null;
  }
}
```

### NotificationService.java
```java
@Service
@RequiredArgsConstructor
public class NotificationService {
    
    private final NotificationRepository notificationRepository;
    private final WebSocketDeliveryService webSocketDeliveryService;
    
    @Transactional
    public void sendNotification(
        Long receiverId,
        NotificationType type,
        String message,
        Long relatedId,
        String relatedType,
        Long senderId,
        String senderName,
        String link
    ) {
        // 1. DB 저장
        Notification notification = Notification.builder()
            .userId(receiverId)
            .notificationType(type)
            .notificationMessage(message)
            .relatedId(relatedId)
            .relatedType(relatedType)
            .senderId(senderId)
            .senderName(senderName)
            .notificationLink(link)
            .isRead(false)
            .createdAt(LocalDateTime.now())
            .build();
        
        Notification saved = notificationRepository.save(notification);
        
        // 2. WebSocket으로 실시간 전송
        try {
            webSocketDeliveryService.sendNotificationToUser(
                receiverId,
                convert(saved)
            );
        } catch (Exception e) {
            log.warn("WebSocket 전송 실패 (사용자 미접속): userId={}", receiverId);
            // DB에는 저장되었으므로 나중에 조회 가능
        }
    }
}
```

### NotificationContext.jsx
```javascript
import React, { createContext, useContext, useState, useEffect } from 'react';
import { connectNotification, disconnectNotification } from '../api/notificationSocket';

const NotificationContext = createContext();

export function NotificationProvider({ children }) {
  const [notifications, setNotifications] = useState([]);
  const [unreadCount, setUnreadCount] = useState(0);
  const { user } = useAuth();
  
  useEffect(() => {
    if (!user) return;
    
    // WebSocket 연결
    connectNotification(
      user.userId,
      (notification) => {
        // 새 알림 수신
        setNotifications(prev => [notification, ...prev]);
        setUnreadCount(prev => prev + 1);
        
        // 토스트 알림 표시
        showToast(notification.notificationMessage);
      },
      () => console.log('알림 WebSocket 연결 성공'),
      (error) => console.error('알림 WebSocket 에러:', error)
    );
    
    return () => {
      disconnectNotification();
    };
  }, [user]);
  
  // 알림 읽음 처리
  const markAsRead = async (notificationId) => {
    await api.patch(`/api/v1/notifications/${notificationId}/read`);
    setNotifications(prev => 
      prev.map(n => n.notificationId === notificationId 
        ? { ...n, isRead: true } 
        : n
      )
    );
    setUnreadCount(prev => Math.max(0, prev - 1));
  };
  
  return (
    <NotificationContext.Provider value={{ 
      notifications, 
      unreadCount, 
      markAsRead 
    }}>
      {children}
    </NotificationContext.Provider>
  );
}

export const useNotification = () => useContext(NotificationContext);
```

## 5️⃣ 이메일 - SendGrid 발송 (Backend)

### SendGridEmailSender.java
```java
@Component
public class SendGridEmailSender {
    
    @Value("${sendgrid.api.key}")
    private String sendgridApiKey;
    
    @Value("${sendgrid.from.email}")
    private String defaultFromEmail;
    
    public Response send(EmailSendRequestDTO dto, List<MultipartFile> attachments) throws IOException {
        SendGrid sg = new SendGrid(sendgridApiKey);
        
        // 1. Mail 객체 생성
        Mail mail = new Mail();
        mail.setFrom(new Email(defaultFromEmail, "CoreConnect"));
        mail.setSubject(dto.getEmailTitle());
        
        // 2. HTML 본문 (HTML + Plain Text)
        Content htmlContent = new Content("text/html", dto.getEmailContent());
        mail.addContent(htmlContent);
        
        // 3. 수신자 추가
        Personalization personalization = new Personalization();
        for (String to : dto.getRecipientAddress()) {
            personalization.addTo(new Email(to));
        }
        
        // CC, BCC
        if (dto.getCcAddresses() != null) {
            for (String cc : dto.getCcAddresses()) {
                personalization.addCc(new Email(cc));
            }
        }
        if (dto.getBccAddresses() != null) {
            for (String bcc : dto.getBccAddresses()) {
                personalization.addBcc(new Email(bcc));
            }
        }
        
        mail.addPersonalization(personalization);
        
        // 4. 첨부파일 처리
        if (attachments != null && !attachments.isEmpty()) {
            for (MultipartFile file : attachments) {
                Attachments sendGridAtt = new Attachments();
                sendGridAtt.setFilename(file.getOriginalFilename());
                sendGridAtt.setType(file.getContentType());
                
                // Base64 인코딩 (SendGrid API 요구사항)
                String encoded = Base64.getEncoder()
                    .encodeToString(file.getBytes());
                sendGridAtt.setContent(encoded);
                
                mail.addAttachments(sendGridAtt);
            }
        }
        
        // 5. SendGrid API 호출
        Request request = new Request();
        request.setMethod(Method.POST);
        request.setEndpoint("mail/send");
        request.setBody(mail.build());
        
        Response response = sg.api(request);
        
        if (response.getStatusCode() >= 200 && response.getStatusCode() < 300) {
            log.info("이메일 발송 성공: statusCode={}", response.getStatusCode());
        } else {
            log.error("이메일 발송 실패: statusCode={}, body={}", 
                response.getStatusCode(), response.getBody());
        }
        
        return response;
    }
}
```

## 6️⃣ WebSocket 인증 (Backend)

### WebSocketAuthInterceptor.java
```java
@Component
@RequiredArgsConstructor
@Slf4j
public class WebSocketAuthInterceptor implements HandshakeInterceptor {
    
    private final JwtUtil jwtUtil;
    
    @Override
    public boolean beforeHandshake(
        ServerHttpRequest request,
        ServerHttpResponse response,
        WebSocketHandler wsHandler,
        Map<String, Object> attributes
    ) throws Exception {
        
        log.info("[WebSocketAuth] Handshake 시작");
        
        // 1. 쿠키에서 JWT 토큰 추출
        if (request instanceof ServletServerHttpRequest) {
            ServletServerHttpRequest servletRequest = 
                (ServletServerHttpRequest) request;
            
            Cookie[] cookies = servletRequest.getServletRequest().getCookies();
            
            if (cookies != null) {
                for (Cookie cookie : cookies) {
                    if ("access_token".equals(cookie.getName())) {
                        String token = cookie.getValue();
                        
                        // 2. 토큰 검증
                        if (jwtUtil.validateToken(token)) {
                            Long userId = jwtUtil.getUserIdFromToken(token);
                            
                            // 3. WebSocket 세션에 userId 저장
                            attributes.put("userId", userId);
                            attributes.put("username", jwtUtil.getUsernameFromToken(token));
                            
                            log.info("[WebSocketAuth] 인증 성공: userId={}", userId);
                            return true;  // 연결 허용
                        }
                    }
                }
            }
        }
        
        log.warn("[WebSocketAuth] 인증 실패: 토큰 없음 또는 유효하지 않음");
        return false;  // 연결 거부
    }
    
    @Override
    public void afterHandshake(...) {
        // Handshake 후 처리 (로깅 등)
    }
}
```

## 7️⃣ Nginx WebSocket 프록시 설정

### nginx.conf
```nginx
# WebSocket 프록시 설정
location /ws {
    proxy_pass http://boot-container:8080;
    
    # HTTP 1.1 필수 (WebSocket은 HTTP/1.1만 지원)
    proxy_http_version 1.1;
    
    # WebSocket Upgrade 헤더 (필수!)
    proxy_set_header Upgrade $http_upgrade;
    proxy_set_header Connection "upgrade";
    
    # 기본 프록시 헤더
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;
    
    # 쿠키 전달 (JWT 인증 필수!)
    proxy_set_header Cookie $http_cookie;
    proxy_pass_request_headers on;
    
    # WebSocket 타임아웃 (24시간)
    proxy_read_timeout 86400s;
    proxy_send_timeout 86400s;
    
    # 버퍼링 비활성화 (실시간 통신)
    proxy_buffering off;
    
    # 연결 재시도
    proxy_connect_timeout 10s;
    proxy_next_upstream error timeout http_502 http_503;
    proxy_next_upstream_tries 3;
}
```

**주의사항**:
- `proxy_http_version 1.1` 없으면 WebSocket 작동 안 함
- `Upgrade`, `Connection` 헤더 필수
- `proxy_buffering off` 권장 (실시간성)

## 8️⃣ 이메일 예약 발송 스케줄러

### EmailReservationScheduler.java
```java
@Component
@RequiredArgsConstructor
@Slf4j
public class EmailReservationScheduler {
    
    private final EmailRepository emailRepository;
    private final SendGridEmailSender sendGridEmailSender;
    private final NotificationService notificationService;
    
    // 1분마다 실행
    @Scheduled(fixedDelay = 60000, initialDelay = 10000)
    public void sendScheduledEmails() {
        log.info("[Scheduler] 예약 이메일 체크 시작");
        
        LocalDateTime now = LocalDateTime.now();
        
        // 발송 시간이 지난 예약 이메일 조회
        List<Email> scheduledEmails = emailRepository
            .findByReservedAtBeforeAndEmailStatus(
                now, 
                EmailStatusEnum.RESERVED
            );
        
        log.info("[Scheduler] 발송 대상: {}건", scheduledEmails.size());
        
        for (Email email : scheduledEmails) {
            try {
                // SendGrid로 발송
                sendGridEmailSender.send(email);
                
                // 상태 업데이트
                email.setEmailStatus(EmailStatusEnum.SENT);
                email.setSentAt(LocalDateTime.now());
                emailRepository.save(email);
                
                // 발신자에게 발송 완료 알림
                notificationService.sendNotification(
                    email.getSenderId(),
                    NotificationType.EMAIL,
                    "[예약 발송 완료] " + email.getEmailTitle(),
                    email.getEmailId(),
                    "EMAIL",
                    null, null, null
                );
                
                log.info("[Scheduler] 발송 성공: emailId={}", email.getEmailId());
                
            } catch (Exception e) {
                log.error("[Scheduler] 발송 실패: emailId={}, error={}", 
                    email.getEmailId(), e.getMessage());
                
                // 실패 상태 업데이트
                email.setEmailStatus(EmailStatusEnum.FAILED);
                emailRepository.save(email);
            }
        }
    }
}
```

**설명**:
- `@Scheduled`: Spring의 스케줄링 기능
- `fixedDelay = 60000`: 이전 실행 완료 후 60초 대기
- `initialDelay = 10000`: 서버 시작 10초 후 첫 실행
- 예약 시간 지난 메일도 자동 발송 (서버 재시작 대응)

```java
@Service
public class S3Service {
    
    private final AmazonS3 amazonS3;
    
    @Value("${aws.s3.bucket}")
    private String bucket;
    
    public String uploadFile(MultipartFile file) throws IOException {
        // 1. 고유 파일명 생성
        String fileName = UUID.randomUUID() + "_" + file.getOriginalFilename();
        String folderPath = "email-attachments/" + LocalDate.now() + "/";
        String key = folderPath + fileName;
        
        // 2. S3 업로드
        ObjectMetadata metadata = new ObjectMetadata();
        metadata.setContentType(file.getContentType());
        metadata.setContentLength(file.getSize());
        
        amazonS3.putObject(
            new PutObjectRequest(bucket, key, file.getInputStream(), metadata)
                .withCannedAcl(CannedAccessControlList.PublicRead)
        );
        
        // 3. URL 반환
        String fileUrl = amazonS3.getUrl(bucket, key).toString();
        
        log.info("S3 업로드 성공: key={}, url={}", key, fileUrl);
        return fileUrl;
    }
    
    // Presigned URL 생성 (보안 강화)
    public String generatePresignedUrl(String fileUrl) {
        String key = extractKeyFromUrl(fileUrl);
        
        Date expiration = new Date();
        expiration.setTime(expiration.getTime() + 1000 * 60 * 60);  // 1시간
        
        GeneratePresignedUrlRequest request = new GeneratePresignedUrlRequest(bucket, key)
            .withMethod(HttpMethod.GET)
            .withExpiration(expiration);
        
        URL url = amazonS3.generatePresignedUrl(request);
        return url.toString();
    }
}
```

## 🔟 JWT 인증 (Backend)

### JwtAuthFilter.java
```java
@Component
@RequiredArgsConstructor
public class JwtAuthFilter extends OncePerRequestFilter {
    
    private final JwtUtil jwtUtil;
    private final CustomUserDetailsService userDetailsService;
    
    @Override
    protected void doFilterInternal(
        HttpServletRequest request,
        HttpServletResponse response,
        FilterChain filterChain
    ) throws ServletException, IOException {
        
        // 1. 쿠키에서 토큰 추출
        String token = null;
        Cookie[] cookies = request.getCookies();
        
        if (cookies != null) {
            for (Cookie cookie : cookies) {
                if ("access_token".equals(cookie.getName())) {
                    token = cookie.getValue();
                    break;
                }
            }
        }
        
        // 2. 토큰 검증 및 인증 설정
        if (token != null && jwtUtil.validateToken(token)) {
            String username = jwtUtil.getUsernameFromToken(token);
            
            UserDetails userDetails = userDetailsService.loadUserByUsername(username);
            
            UsernamePasswordAuthenticationToken authentication = 
                new UsernamePasswordAuthenticationToken(
                    userDetails, null, userDetails.getAuthorities()
                );
            
            authentication.setDetails(
                new WebAuthenticationDetailsSource().buildDetails(request)
            );
            
            // SecurityContext에 인증 정보 설정
            SecurityContextHolder.getContext().setAuthentication(authentication);
            
            log.debug("[JWT] 인증 성공: username={}, URI={}", 
                username, request.getRequestURI());
        }
        
        filterChain.doFilter(request, response);
    }
}
```

## 🔑 핵심 개념 요약

### WebSocket vs HTTP
| 구분 | WebSocket | HTTP |
|------|-----------|------|
| 연결 | 지속적 (persistent) | 요청/응답 후 종료 |
| 방향 | 양방향 (full-duplex) | 단방향 (client → server) |
| 오버헤드 | 낮음 (헤더 최소) | 높음 (매번 헤더 전송) |
| 실시간성 | 즉시 | 폴링 주기만큼 지연 |
| 사용 예 | 채팅, 알림, 게임 | RESTful API, 웹페이지 |

### STOMP 프로토콜
- **Simple Text Oriented Messaging Protocol**
- **Pub/Sub 패턴**: Publisher가 토픽에 발행, Subscriber가 구독
- **Destination**: `/app` (send), `/topic` (subscribe), `/queue` (point-to-point)
- **프레임**: CONNECT, SUBSCRIBE, SEND, MESSAGE, DISCONNECT

### JWT (JSON Web Token)
- **구조**: Header.Payload.Signature
- **Stateless**: 서버에 세션 저장 불필요
- **HttpOnly Cookie**: XSS 공격 방지
- **만료 시간**: 보통 1-24시간
