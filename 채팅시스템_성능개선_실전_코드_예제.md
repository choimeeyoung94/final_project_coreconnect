### 1.1 WebSocket Handler (고급)

```java
// ChatWebSocketHandler.java
@Component
@RequiredArgsConstructor
@Slf4j
public class ChatWebSocketHandler extends TextWebSocketHandler {
    
    private final WebSocketSessionManager sessionManager;
    private final ChatMessagePublisher messagePublisher;
    private final MeterRegistry meterRegistry;
    
    // ⭐ 연결 수 메트릭
    private final AtomicInteger activeConnections = new AtomicInteger(0);
    
    @PostConstruct
    public void init() {
        Gauge.builder("websocket.active.connections", activeConnections, AtomicInteger::get)
                .description("Active WebSocket connections")
                .register(meterRegistry);
    }
    
    @Override
    public void afterConnectionEstablished(WebSocketSession session) throws Exception {
        try {
            // 1. 사용자 인증 정보 추출
            Integer userId = extractUserId(session);
            Integer roomId = extractRoomId(session);
            
            if (userId == null || roomId == null) {
                log.warn("[WebSocket] 인증 실패 - sessionId: {}", session.getId());
                session.close(CloseStatus.NOT_ACCEPTABLE.withReason("Authentication failed"));
                return;
            }
            
            // 2. ⭐ 세션 등록
            sessionManager.addSession(userId, roomId, session.getId(), session);
            
            // 3. ⭐ 메트릭 증가
            activeConnections.incrementAndGet();
            
            // 4. ⭐ 입장 알림 (선택사항)
            ChatMessage joinMessage = ChatMessage.builder()
                    .roomId(roomId)
                    .senderId(userId)
                    .type(MessageType.JOIN)
                    .content(userId + "님이 입장했습니다.")
                    .timestamp(System.currentTimeMillis())
                    .build();
            
            messagePublisher.publishMessage(roomId, joinMessage);
            
            log.info("[WebSocket] 연결 성공 - userId: {}, roomId: {}, sessionId: {}, total: {}", 
                    userId, roomId, session.getId(), activeConnections.get());
            
        } catch (Exception e) {
            log.error("[WebSocket] 연결 처리 실패", e);
            session.close(CloseStatus.SERVER_ERROR);
        }
    }
    
    @Override
    protected void handleTextMessage(WebSocketSession session, TextMessage message) throws Exception {
        try {
            long startTime = System.currentTimeMillis();
            
            // 1. 메시지 파싱
            ChatMessage chatMessage = objectMapper.readValue(message.getPayload(), ChatMessage.class);
            
            // 2. ⭐ 메시지 검증
            if (!validateMessage(chatMessage)) {
                log.warn("[WebSocket] 메시지 검증 실패 - sessionId: {}", session.getId());
                return;
            }
            
            // 3. ⭐ Rate Limiting (초당 10개 제한)
            if (!checkRateLimit(chatMessage.getSenderId())) {
                log.warn("[WebSocket] Rate limit 초과 - userId: {}", chatMessage.getSenderId());
                session.sendMessage(new TextMessage("{\"error\":\"Rate limit exceeded\"}"));
                return;
            }
            
            // 4. ⭐ 비동기 발행
            messagePublisher.publishMessageAsync(chatMessage.getRoomId(), chatMessage);
            
            // 5. ⭐ 메트릭 기록
            long latency = System.currentTimeMillis() - startTime;
            meterRegistry.timer("websocket.message.processing.time")
                    .record(latency, TimeUnit.MILLISECONDS);
            
            log.debug("[WebSocket] 메시지 처리 완료 - userId: {}, latency: {}ms", 
                    chatMessage.getSenderId(), latency);
            
        } catch (Exception e) {
            log.error("[WebSocket] 메시지 처리 실패", e);
            session.sendMessage(new TextMessage("{\"error\":\"Message processing failed\"}"));
        }
    }
    
    @Override
    public void afterConnectionClosed(WebSocketSession session, CloseStatus status) throws Exception {
        try {
            Integer userId = extractUserId(session);
            Integer roomId = extractRoomId(session);
            
            if (userId != null && roomId != null) {
                // 1. ⭐ 세션 제거
                sessionManager.removeSession(userId, roomId);
                
                // 2. ⭐ 메트릭 감소
                activeConnections.decrementAndGet();
                
                // 3. ⭐ 퇴장 알림
                ChatMessage leaveMessage = ChatMessage.builder()
                        .roomId(roomId)
                        .senderId(userId)
                        .type(MessageType.LEAVE)
                        .content(userId + "님이 퇴장했습니다.")
                        .timestamp(System.currentTimeMillis())
                        .build();
                
                messagePublisher.publishMessage(roomId, leaveMessage);
                
                log.info("[WebSocket] 연결 종료 - userId: {}, roomId: {}, status: {}, total: {}", 
                        userId, roomId, status, activeConnections.get());
            }
            
        } catch (Exception e) {
            log.error("[WebSocket] 연결 종료 처리 실패", e);
        }
    }
    
    @Override
    public void handleTransportError(WebSocketSession session, Throwable exception) throws Exception {
        log.error("[WebSocket] 전송 에러 - sessionId: {}", session.getId(), exception);
        
        // ⭐ 에러 메트릭 증가
        meterRegistry.counter("websocket.errors").increment();
        
        // 연결 종료
        if (session.isOpen()) {
            session.close(CloseStatus.SERVER_ERROR);
        }
    }
    
    /**
     * ⭐ Rate Limiting (Guava RateLimiter 사용)
     */
    private final Map<Integer, RateLimiter> rateLimiters = new ConcurrentHashMap<>();
    
    private boolean checkRateLimit(Integer userId) {
        RateLimiter limiter = rateLimiters.computeIfAbsent(
            userId, 
            k -> RateLimiter.create(10.0)  // 초당 10개
        );
        return limiter.tryAcquire();
    }
    
    /**
     * ⭐ 메시지 검증
     */
    private boolean validateMessage(ChatMessage message) {
        if (message == null) return false;
        if (message.getRoomId() == null) return false;
        if (message.getSenderId() == null) return false;
        if (message.getContent() == null || message.getContent().isEmpty()) return false;
        if (message.getContent().length() > 10000) return false;  // 10KB 제한
        return true;
    }
    
    /**
     * ⭐ 세션에서 userId 추출
     */
    private Integer extractUserId(WebSocketSession session) {
        Map<String, Object> attributes = session.getAttributes();
        return (Integer) attributes.get("userId");
    }
    
    /**
     * ⭐ 세션에서 roomId 추출
     */
    private Integer extractRoomId(WebSocketSession session) {
        Map<String, Object> attributes = session.getAttributes();
        return (Integer) attributes.get("roomId");
    }
}
```

```java
// WebSocketAuthInterceptor.java
@Component
@RequiredArgsConstructor
@Slf4j
public class WebSocketAuthInterceptor implements HandshakeInterceptor {
    
    private final JwtTokenProvider jwtTokenProvider;
    private final UserRepository userRepository;
    
    @Override
    public boolean beforeHandshake(
            ServerHttpRequest request, 
            ServerHttpResponse response,
            WebSocketHandler wsHandler, 
            Map<String, Object> attributes) throws Exception {
        
        try {
            // 1. ⭐ JWT 토큰 추출
            String token = extractToken(request);
            if (token == null) {
                log.warn("[WebSocket] 토큰 없음 - URI: {}", request.getURI());
                return false;
            }
            
            // 2. ⭐ 토큰 검증
            if (!jwtTokenProvider.validateToken(token)) {
                log.warn("[WebSocket] 토큰 검증 실패 - token: {}", token);
                return false;
            }
            
            // 3. ⭐ 사용자 정보 추출
            Integer userId = jwtTokenProvider.getUserIdFromToken(token);
            User user = userRepository.findById(userId)
                    .orElseThrow(() -> new IllegalArgumentException("사용자 없음: " + userId));
            
            // 4. ⭐ roomId 추출 (쿼리 파라미터)
            Integer roomId = extractRoomId(request);
            if (roomId == null) {
                log.warn("[WebSocket] roomId 없음 - URI: {}", request.getURI());
                return false;
            }
            
            // 5. ⭐ 세션 속성에 저장
            attributes.put("userId", userId);
            attributes.put("userName", user.getName());
            attributes.put("userEmail", user.getEmail());
            attributes.put("roomId", roomId);
            attributes.put("connectedAt", System.currentTimeMillis());
            
            log.info("[WebSocket] 인증 성공 - userId: {}, roomId: {}", userId, roomId);
            return true;
            
        } catch (Exception e) {
            log.error("[WebSocket] 인증 실패", e);
            return false;
        }
    }
    
    @Override
    public void afterHandshake(
            ServerHttpRequest request, 
            ServerHttpResponse response,
            WebSocketHandler wsHandler, 
            Exception exception) {
        
        if (exception != null) {
            log.error("[WebSocket] Handshake 에러", exception);
        }
    }
    
    /**
     * ⭐ JWT 토큰 추출 (쿼리 파라미터 또는 헤더)
     */
    private String extractToken(ServerHttpRequest request) {
        // 1. 쿼리 파라미터에서 추출
        String query = request.getURI().getQuery();
        if (query != null && query.contains("token=")) {
            String[] params = query.split("&");
            for (String param : params) {
                if (param.startsWith("token=")) {
                    return param.substring(6);
                }
            }
        }
        
        // 2. 헤더에서 추출
        HttpHeaders headers = request.getHeaders();
        String authorization = headers.getFirst("Authorization");
        if (authorization != null && authorization.startsWith("Bearer ")) {
            return authorization.substring(7);
        }
        
        return null;
    }
    
    /**
     * ⭐ roomId 추출
     */
    private Integer extractRoomId(ServerHttpRequest request) {
        String query = request.getURI().getQuery();
        if (query != null && query.contains("roomId=")) {
            String[] params = query.split("&");
            for (String param : params) {
                if (param.startsWith("roomId=")) {
                    try {
                        return Integer.parseInt(param.substring(7));
                    } catch (NumberFormatException e) {
                        log.warn("[WebSocket] roomId 파싱 실패: {}", param);
                    }
                }
            }
        }
        return null;
    }
}
```

## 2. 메시지 배치 처리

### 2.1 Batch Processor (고성능)

```java
// MessageBatchProcessor.java
@Component
@RequiredArgsConstructor
@Slf4j
public class MessageBatchProcessor {
    
    private final ChatMessagePublisher messagePublisher;
    private final ChatRepository chatRepository;
    private final MeterRegistry meterRegistry;
    
    // ⭐ 메시지 큐 (동시성 안전)
    private final BlockingQueue<ChatMessage> messageQueue = new LinkedBlockingQueue<>(10000);
    
    // ⭐ 배치 설정
    private static final int BATCH_SIZE = 100;           // 한 번에 100개
    private static final int BATCH_INTERVAL_MS = 10;     // 10ms마다 처리
    private static final int THREAD_COUNT = 4;           // 처리 스레드 4개
    
    private final ScheduledExecutorService scheduler = Executors.newScheduledThreadPool(THREAD_COUNT);
    private final ExecutorService executor = Executors.newFixedThreadPool(THREAD_COUNT);
    
    @PostConstruct
    public void start() {
        log.info("[BatchProcessor] 시작 - threads: {}, batchSize: {}, interval: {}ms", 
                THREAD_COUNT, BATCH_SIZE, BATCH_INTERVAL_MS);
        
        // ⭐ 배치 처리 스케줄러
        for (int i = 0; i < THREAD_COUNT; i++) {
            final int threadId = i;
            scheduler.scheduleAtFixedRate(
                () -> processBatch(threadId),
                0,
                BATCH_INTERVAL_MS,
                TimeUnit.MILLISECONDS
            );
        }
        
        // ⭐ 큐 크기 모니터링
        Gauge.builder("message.queue.size", messageQueue, Queue::size)
                .description("Message queue size")
                .register(meterRegistry);
    }
    
    @PreDestroy
    public void stop() {
        log.info("[BatchProcessor] 종료 중...");
        scheduler.shutdown();
        executor.shutdown();
        try {
            if (!scheduler.awaitTermination(10, TimeUnit.SECONDS)) {
                scheduler.shutdownNow();
            }
            if (!executor.awaitTermination(10, TimeUnit.SECONDS)) {
                executor.shutdownNow();
            }
        } catch (InterruptedException e) {
            scheduler.shutdownNow();
            executor.shutdownNow();
            Thread.currentThread().interrupt();
        }
        log.info("[BatchProcessor] 종료 완료");
    }
    
    /**
     * ⭐ 메시지 추가 (비동기)
     */
    public boolean addMessage(ChatMessage message) {
        boolean added = messageQueue.offer(message);
        if (!added) {
            log.warn("[BatchProcessor] 큐 가득참 - messageId: {}", message.getId());
            meterRegistry.counter("message.queue.full").increment();
        }
        return added;
    }
    
    /**
     * ⭐ 배치 처리 (핵심 로직)
     */
    private void processBatch(int threadId) {
        try {
            List<ChatMessage> batch = new ArrayList<>();
            
            // 1. 큐에서 메시지 꺼내기
            messageQueue.drainTo(batch, BATCH_SIZE);
            
            if (batch.isEmpty()) {
                return;
            }
            
            long startTime = System.currentTimeMillis();
            
            // 2. ⭐ 채팅방별로 그룹화 (효율성 향상)
            Map<Integer, List<ChatMessage>> grouped = batch.stream()
                    .collect(Collectors.groupingBy(ChatMessage::getRoomId));
            
            // 3. ⭐ 병렬 처리
            grouped.entrySet().parallelStream().forEach(entry -> {
                Integer roomId = entry.getKey();
                List<ChatMessage> messages = entry.getValue();
                
                try {
                    // Redis Pub/Sub로 배치 발행
                    messagePublisher.publishBatch(roomId, messages);
                    
                    // DB에 배치 저장
                    saveBatch(messages);
                    
                } catch (Exception e) {
                    log.error("[BatchProcessor] 배치 처리 실패 - roomId: {}", roomId, e);
                    meterRegistry.counter("message.batch.errors").increment();
                }
            });
            
            long latency = System.currentTimeMillis() - startTime;
            
            // 4. ⭐ 메트릭 기록
            meterRegistry.counter("message.batch.processed").increment(batch.size());
            meterRegistry.timer("message.batch.processing.time")
                    .record(latency, TimeUnit.MILLISECONDS);
            
            log.debug("[BatchProcessor] 배치 처리 완료 - thread: {}, count: {}, latency: {}ms", 
                    threadId, batch.size(), latency);
            
        } catch (Exception e) {
            log.error("[BatchProcessor] 배치 처리 예외 - thread: {}", threadId, e);
        }
    }
    
    /**
     * ⭐ DB 배치 저장 (JPA Batch Insert)
     */
    @Transactional
    private void saveBatch(List<ChatMessage> messages) {
        // JPA Batch Insert 사용
        // application.yml: spring.jpa.properties.hibernate.jdbc.batch_size=100
        
        List<Chat> entities = messages.stream()
                .map(this::toEntity)
                .collect(Collectors.toList());
        
        chatRepository.saveAll(entities);
    }
    
    private Chat toEntity(ChatMessage message) {
        // ChatMessage → Chat 변환
        return Chat.builder()
                .id(message.getId())
                .roomId(message.getRoomId())
                .senderId(message.getSenderId())
                .messageContent(message.getContent())
                .sendAt(LocalDateTime.now())
                .build();
    }
}
```

## 3. Redis 캐싱 전략

### 3.1 최근 메시지 캐싱

```java
// ChatCacheService.java
@Service
@RequiredArgsConstructor
@Slf4j
public class ChatCacheService {
    
    private final RedisTemplate<String, Object> redisTemplate;
    private final ChatRepository chatRepository;
    
    private static final int CACHE_SIZE = 100;  // 최근 100개
    private static final int CACHE_TTL_HOURS = 24;
    
    /**
     * ⭐ 최근 메시지 조회 (캐시 우선)
     */
    public List<ChatMessage> getRecentMessages(Integer roomId, int count) {
        String key = "chat:room:" + roomId + ":recent";
        
        try {
            // 1. ⭐ 캐시 조회
            ZSetOperations<String, Object> zset = redisTemplate.opsForZSet();
            Set<Object> cached = zset.reverseRange(key, 0, count - 1);
            
            if (cached != null && !cached.isEmpty()) {
                log.debug("[Cache] HIT - roomId: {}, count: {}", roomId, cached.size());
                return cached.stream()
                        .map(obj -> (ChatMessage) obj)
                        .collect(Collectors.toList());
            }
            
            // 2. ⭐ 캐시 미스 → DB 조회
            log.debug("[Cache] MISS - roomId: {}", roomId);
            List<Chat> messages = chatRepository.findTop100ByRoomIdOrderBySendAtDesc(roomId);
            
            // 3. ⭐ 캐시 저장
            cacheMessages(roomId, messages);
            
            return messages.stream()
                    .map(this::toDto)
                    .collect(Collectors.toList());
            
        } catch (Exception e) {
            log.error("[Cache] 조회 실패 - roomId: {}", roomId, e);
            // ⭐ 캐시 실패 시 DB 직접 조회
            return chatRepository.findTop100ByRoomIdOrderBySendAtDesc(roomId)
                    .stream()
                    .map(this::toDto)
                    .collect(Collectors.toList());
        }
    }
    
    /**
     * ⭐ 메시지 캐시 저장
     */
    public void cacheMessage(Integer roomId, ChatMessage message) {
        String key = "chat:room:" + roomId + ":recent";
        
        try {
            ZSetOperations<String, Object> zset = redisTemplate.opsForZSet();
            
            // 1. ⭐ Sorted Set에 추가 (timestamp를 score로)
            double score = message.getTimestamp();
            zset.add(key, message, score);
            
            // 2. ⭐ 최근 100개만 유지
            Long size = zset.size(key);
            if (size != null && size > CACHE_SIZE) {
                zset.removeRange(key, 0, size - CACHE_SIZE - 1);
            }
            
            // 3. ⭐ TTL 갱신
            redisTemplate.expire(key, CACHE_TTL_HOURS, TimeUnit.HOURS);
            
            log.debug("[Cache] 메시지 저장 - roomId: {}, messageId: {}", roomId, message.getId());
            
        } catch (Exception e) {
            log.error("[Cache] 저장 실패 - roomId: {}", roomId, e);
        }
    }
    
    /**
     * ⭐ 배치 캐시 저장
     */
    private void cacheMessages(Integer roomId, List<Chat> messages) {
        String key = "chat:room:" + roomId + ":recent";
        
        try {
            ZSetOperations<String, Object> zset = redisTemplate.opsForZSet();
            
            // ⭐ Pipeline으로 한 번에 저장
            redisTemplate.executePipelined((RedisCallback<Object>) connection -> {
                messages.forEach(msg -> {
                    ChatMessage dto = toDto(msg);
                    double score = dto.getTimestamp();
                    zset.add(key, dto, score);
                });
                return null;
            });
            
            // TTL 설정
            redisTemplate.expire(key, CACHE_TTL_HOURS, TimeUnit.HOURS);
            
            log.debug("[Cache] 배치 저장 완료 - roomId: {}, count: {}", roomId, messages.size());
            
        } catch (Exception e) {
            log.error("[Cache] 배치 저장 실패 - roomId: {}", roomId, e);
        }
    }
    
    /**
     * ⭐ 읽음 상태 캐싱 (Bitmap 사용)
     */
    public void cacheReadStatus(Integer roomId, Integer userId, Integer messageId) {
        String key = String.format("chat:read:%d:%d", roomId, userId);
        
        try {
            // ⭐ Bitmap으로 매우 효율적으로 저장 (메모리 절약)
            redisTemplate.opsForValue().setBit(key, messageId, true);
            
            // TTL 설정 (7일)
            redisTemplate.expire(key, 7, TimeUnit.DAYS);
            
            log.debug("[Cache] 읽음 상태 저장 - roomId: {}, userId: {}, messageId: {}", 
                    roomId, userId, messageId);
            
        } catch (Exception e) {
            log.error("[Cache] 읽음 상태 저장 실패", e);
        }
    }
    
    /**
     * ⭐ 읽음 상태 조회
     */
    public boolean isMessageRead(Integer roomId, Integer userId, Integer messageId) {
        String key = String.format("chat:read:%d:%d", roomId, userId);
        
        try {
            Boolean isRead = redisTemplate.opsForValue().getBit(key, messageId);
            return Boolean.TRUE.equals(isRead);
        } catch (Exception e) {
            log.error("[Cache] 읽음 상태 조회 실패", e);
            return false;
        }
    }
    
    /**
     * ⭐ 안읽은 메시지 수 조회 (HyperLogLog 사용)
     */
    public long getUnreadCount(Integer roomId, Integer userId) {
        String key = String.format("chat:unread:%d:%d", roomId, userId);
        
        try {
            HyperLogLogOperations<String, Object> hll = redisTemplate.opsForHyperLogLog();
            Long size = hll.size(key);
            return size != null ? size : 0L;
        } catch (Exception e) {
            log.error("[Cache] 안읽은 메시지 수 조회 실패", e);
            return 0L;
        }
    }
    
    private ChatMessage toDto(Chat entity) {
        return ChatMessage.builder()
                .id(entity.getId())
                .roomId(entity.getRoomId())
                .senderId(entity.getSenderId())
                .content(entity.getMessageContent())
                .timestamp(entity.getSendAt().toEpochSecond(ZoneOffset.UTC) * 1000)
                .build();
    }
}
```

```java
// ChatMessageKafkaConsumer.java
@Component
@RequiredArgsConstructor
@Slf4j
public class ChatMessageKafkaConsumer {
    
    private final ChatRepository chatRepository;
    private final ChatCacheService cacheService;
    private final MeterRegistry meterRegistry;
    
    /**
     * ⭐ Kafka 메시지 수신 (배치 처리)
     */
    @KafkaListener(
        topics = "chat-messages",
        groupId = "chat-consumer-group",
        containerFactory = "kafkaListenerContainerFactory",
        concurrency = "10"  // ⭐ 10개 스레드로 병렬 처리
    )
    public void consumeMessages(
            List<ConsumerRecord<String, ChatMessage>> records,
            Acknowledgment acknowledgment) {
        
        long startTime = System.currentTimeMillis();
        
        try {
            if (records.isEmpty()) {
                return;
            }
            
            log.info("[KafkaConsumer] 배치 수신 시작 - count: {}", records.size());
            
            // 1. ⭐ 채팅방별로 그룹화
            Map<Integer, List<ChatMessage>> grouped = records.stream()
                    .map(ConsumerRecord::value)
                    .collect(Collectors.groupingBy(ChatMessage::getRoomId));
            
            // 2. ⭐ 병렬 처리
            grouped.entrySet().parallelStream().forEach(entry -> {
                Integer roomId = entry.getKey();
                List<ChatMessage> messages = entry.getValue();
                
                try {
                    // DB 저장
                    saveBatch(messages);
                    
                    // 캐시 갱신
                    messages.forEach(msg -> cacheService.cacheMessage(roomId, msg));
                    
                    // 읽음 상태 초기화 (비동기)
                    initializeReadStatusAsync(messages);
                    
                    log.debug("[KafkaConsumer] 처리 완료 - roomId: {}, count: {}", 
                            roomId, messages.size());
                    
                } catch (Exception e) {
                    log.error("[KafkaConsumer] 처리 실패 - roomId: {}", roomId, e);
                    meterRegistry.counter("kafka.consumer.errors").increment();
                }
            });
            
            // 3. ⭐ 수동 커밋
            acknowledgment.acknowledge();
            
            long latency = System.currentTimeMillis() - startTime;
            
            // 4. ⭐ 메트릭 기록
            meterRegistry.counter("kafka.messages.consumed").increment(records.size());
            meterRegistry.timer("kafka.batch.processing.time")
                    .record(latency, TimeUnit.MILLISECONDS);
            
            log.info("[KafkaConsumer] 배치 처리 완료 - count: {}, latency: {}ms", 
                    records.size(), latency);
            
        } catch (Exception e) {
            log.error("[KafkaConsumer] 배치 처리 예외", e);
            // ⭐ 예외 발생 시 커밋하지 않음 → 재처리
        }
    }
    
    @Transactional
    private void saveBatch(List<ChatMessage> messages) {
        List<Chat> entities = messages.stream()
                .map(this::toEntity)
                .collect(Collectors.toList());
        
        chatRepository.saveAll(entities);
    }
    
    @Async
    private CompletableFuture<Void> initializeReadStatusAsync(List<ChatMessage> messages) {
        return CompletableFuture.runAsync(() -> {
            // 읽음 상태 초기화 로직
            // ...
        });
    }
    
    private Chat toEntity(ChatMessage message) {
        return Chat.builder()
                .id(message.getId())
                .roomId(message.getRoomId())
                .senderId(message.getSenderId())
                .messageContent(message.getContent())
                .sendAt(LocalDateTime.now())
                .build();
    }
```

```java
// ChatMetricsService.java
@Service
@RequiredArgsConstructor
@Slf4j
public class ChatMetricsService {
    
    private final MeterRegistry meterRegistry;
    private final WebSocketSessionManager sessionManager;
    
    // ⭐ 메트릭 컬렉터
    private final ConcurrentHashMap<String, AtomicLong> counters = new ConcurrentHashMap<>();
    private final ConcurrentHashMap<String, Timer> timers = new ConcurrentHashMap<>();
    
    @PostConstruct
    public void init() {
        registerMetrics();
        startPeriodicReporting();
    }
    
    /**
     * ⭐ 메트릭 등록
     */
    private void registerMetrics() {
        // 1. WebSocket 연결 수
        Gauge.builder("websocket.connections.active", sessionManager, 
                mgr -> mgr.getStats().get("totalUsers"))
                .description("Active WebSocket connections")
                .register(meterRegistry);
        
        // 2. 채팅방 수
        Gauge.builder("chat.rooms.active", sessionManager,
                mgr -> mgr.getStats().get("totalRooms"))
                .description("Active chat rooms")
                .register(meterRegistry);
        
        // 3. 메시지 처리 TPS
        Counter.builder("chat.messages.sent.total")
                .description("Total messages sent")
                .register(meterRegistry);
        
        // 4. 메시지 지연 시간
        Timer.builder("chat.message.latency")
                .description("Message end-to-end latency")
                .publishPercentiles(0.5, 0.95, 0.99)  // P50, P95, P99
                .register(meterRegistry);
        
        // 5. 에러율
        Counter.builder("chat.errors.total")
                .description("Total errors")
                .register(meterRegistry);
    }
    
    /**
     * ⭐ 주기적인 메트릭 리포팅
     */
    @Scheduled(fixedRate = 10000)  // 10초마다
    private void startPeriodicReporting() {
        try {
            Map<String, Object> stats = sessionManager.getStats();
            
            log.info("""
                [Metrics] 현재 상태
                - WebSocket 연결: {}
                - 활성 채팅방: {}
                - 채팅방별 연결: {}
                """,
                stats.get("totalUsers"),
                stats.get("totalRooms"),
                stats.get("roomDetails")
            );
            
        } catch (Exception e) {
            log.error("[Metrics] 리포팅 실패", e);
        }
    }
    
    /**
     * ⭐ 메시지 지연 시간 기록
     */
    public void recordMessageLatency(long latencyMs) {
        meterRegistry.timer("chat.message.latency")
                .record(latencyMs, TimeUnit.MILLISECONDS);
    }
    
    /**
     * ⭐ 메시지 전송 카운트 증가
     */
    public void incrementMessageSent() {
        meterRegistry.counter("chat.messages.sent.total").increment();
    }
    
    /**
     * ⭐ 에러 카운트 증가
     */
    public void incrementError(String errorType) {
        meterRegistry.counter("chat.errors.total", "type", errorType).increment();
    }
```

```javascript
// k6-chat-load-test.js
import ws from 'k6/ws';
import { check, sleep } from 'k6';
import { Counter, Trend, Gauge } from 'k6/metrics';

// ⭐ 커스텀 메트릭
const wsConnections = new Gauge('websocket_connections');
const messagesSent = new Counter('messages_sent');
const messagesReceived = new Counter('messages_received');
const messageLatency = new Trend('message_latency');

// ⭐ 테스트 설정
export const options = {
    stages: [
        { duration: '2m', target: 1000 },    // 1,000명까지 램프업
        { duration: '3m', target: 5000 },    // 5,000명까지 램프업
        { duration: '5m', target: 10000 },   // 10,000명까지 램프업
        { duration: '10m', target: 10000 },  // 10,000명 유지
        { duration: '2m', target: 0 },       // 램프다운
    ],
    thresholds: {
        // ⭐ 성능 목표
        'websocket_connections': ['value>0'],
        'message_latency': ['p(95)<200', 'p(99)<500'],  // P95<200ms, P99<500ms
        'messages_sent': ['count>100000'],               // 최소 10만개
        'checks': ['rate>0.99'],                         // 99% 성공률
    },
};

// ⭐ 테스트 시나리오
export default function () {
    const userId = __VU;  // Virtual User ID
    const roomId = Math.floor(Math.random() * 10) + 1;  // 1-10번 채팅방
    const token = __ENV.JWT_TOKEN || 'test-token';
    
    const url = `ws://localhost:8080/ws/chat?token=${token}&roomId=${roomId}`;
    
    const params = {
        tags: { name: 'ChatWebSocket' },
    };
    
    const res = ws.connect(url, params, function (socket) {
        wsConnections.add(1);
        
        socket.on('open', () => {
            console.log(`[User ${userId}] 연결 성공 - roomId: ${roomId}`);
            
            // ⭐ 메시지 전송 (10초마다)
            socket.setInterval(() => {
                const sendTime = Date.now();
                const message = JSON.stringify({
                    roomId: roomId,
                    content: `Test message from user ${userId} at ${sendTime}`,
                    timestamp: sendTime
                });
                
                socket.send(message);
                messagesSent.add(1);
                
                // 메시지 속성에 전송 시간 저장
                socket.lastSendTime = sendTime;
            }, 10000);
        });
        
        socket.on('message', (data) => {
            messagesReceived.add(1);
            
            try {
                const message = JSON.parse(data);
                const receiveTime = Date.now();
                
                // ⭐ 지연 시간 계산
                if (socket.lastSendTime) {
                    const latency = receiveTime - socket.lastSendTime;
                    messageLatency.add(latency);
                    
                    if (latency > 1000) {
                        console.log(`[User ${userId}] 높은 지연! ${latency}ms`);
                    }
                }
            } catch (e) {
                console.error(`[User ${userId}] 메시지 파싱 실패:`, e);
            }
        });
        
        socket.on('error', (e) => {
            console.error(`[User ${userId}] WebSocket 에러:`, e);
        });
        
        socket.on('close', () => {
            console.log(`[User ${userId}] 연결 종료`);
            wsConnections.add(-1);
        });
        
        // ⭐ 10분간 연결 유지
        socket.setTimeout(() => {
            console.log(`[User ${userId}] 타임아웃, 연결 종료`);
            socket.close();
        }, 600000);
    });
    
    check(res, {
        'status is 101': (r) => r && r.status === 101,
    });
    
    // 다음 사용자 연결 전 대기 (램프업 속도 조절)
    sleep(0.1);
}

// ⭐ 테스트 종료 후 요약
export function handleSummary(data) {
    return {
        'summary.json': JSON.stringify(data),
        stdout: textSummary(data, { indent: ' ', enableColors: true }),
    };
}
```
